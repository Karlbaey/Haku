{{- $page := .page -}} {{- $value := .value | default $page.Params.originalDate -}} {{- $display := "" -}} {{- $datetime := "" -}} {{- with $value -}} {{- if eq (printf "%T" .) "time.Time" -}} {{- $display = partial "format-date.html" . -}} {{- $datetime = .Format "2006-01-02" -}} {{- else -}} {{- $display = strings.TrimSpace (printf "%v" .) -}} {{- $datetime = $display -}} {{- end -}} {{- end -}} {{- if not $display -}} {{- $display = partial "format-date.html" $page.Date -}} {{- $datetime = $page.Date.Format "2006-01-02" -}} {{- end -}} {{- dict "display" $display "datetime" $datetime | jsonify -}}
